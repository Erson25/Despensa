/** * DESPENSA DEL CAMPO — receptor de pedidos + catálogo * * Esta hoja de cálculo es el back de la vitrina. Hace tres cosas: * 1. Guarda los pedidos que llegan (hojas "Pedidos" y "Lineas"). * 2. Consolida los kilos por producto para la compra (hoja "Consolidado"). * 3. Le entrega a la vitrina el catálogo y la configuración (hojas "Catalogo" y "Config"). * * PRIMER PASO, UNA SOLA VEZ: * Menú Despensa → "Crear hojas base". Eso arma Catalogo y Config con ejemplos. */ var HOJA_PEDIDOS = 'Pedidos'; var HOJA_LINEAS = 'Lineas'; var HOJA_CONSOLIDADO = 'Consolidado'; var HOJA_CATALOGO = 'Catalogo'; var HOJA_CONFIG = 'Config'; var ENC_PEDIDOS = ['Marca temporal', 'Jornada', 'Cliente', 'WhatsApp', 'Punto de recogida', 'Franja', 'Kg totales', 'Total $', 'Detalle del pedido', 'Notas', 'Estado', 'Pagado']; var ENC_LINEAS = ['Marca temporal', 'Jornada', 'Cliente', 'Producto', 'Kg', 'Precio $/kg', 'Subtotal $']; var ENC_CATALOGO = ['Orden', 'Producto', 'Precio venta $/kg', 'Precio de referencia $/kg', 'Mostrar ahorro', 'Paso (kg)', 'Descripción corta', 'Archivo de foto', 'Disponible']; /* ================================================================ 1. LO QUE LA VITRINA PIDE (catálogo y configuración) ================================================================ */ function doGet(e) { var accion = (e && e.parameter && e.parameter.accion) || 'catalogo'; if (accion !== 'catalogo') return json({ ok: true, mensaje: 'Receptor activo' }); try { return json({ ok: true, productos: leerCatalogo(), config: leerConfig() }); } catch (err) { return json({ ok: false, error: String(err) }); } } function leerCatalogo() { var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJA_CATALOGO); if (!hoja || hoja.getLastRow() < 2) return []; var filas = hoja.getRange(2, 1, hoja.getLastRow() - 1, ENC_CATALOGO.length).getValues(); var lista = []; filas.forEach(function (f) { var nombre = String(f[1]).trim(); if (!nombre) return; // fila vacía: se ignora var precio = Number(f[2]) || 0; if (precio <= 0) return; // sin precio no se publica lista.push({ orden: Number(f[0]) || 999, nombre: nombre, precio: precio, plaza: Number(f[3]) || 0, ahorro: normalizar(f[4]) || 'auto', // 'si' | 'no' | 'auto' paso: Number(f[5]) || 1, nota: String(f[6] || '').trim(), foto: String(f[7] || '').trim(), hay: normalizar(f[8]) !== 'no' }); }); lista.sort(function (a, b) { return a.orden - b.orden; }); return lista; } function leerConfig() { var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJA_CONFIG); var cfg = { puntos: [], franjas: [], minimo: 0, titular: '', mensaje: '' }; if (!hoja || hoja.getLastRow() < 2) return cfg; hoja.getRange(2, 1, hoja.getLastRow() - 1, 2).getValues().forEach(function (f) { var clave = normalizar(f[0]), valor = String(f[1] || '').trim(); if (!clave || !valor) return; if (clave === 'punto de recogida') cfg.puntos.push(valor); else if (clave === 'franja de entrega') cfg.franjas.push(valor); else if (clave === 'pedido minimo') cfg.minimo = Number(String(valor).replace(/\D/g, '')) || 0; else if (clave === 'titular') cfg.titular = valor; else if (clave === 'mensaje de la semana') cfg.mensaje = valor; }); return cfg; } function normalizar(v) { return String(v || '').trim().toLowerCase() .normalize('NFD').replace(/[\u0300-\u036f]/g, ''); // quita tildes } function json(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /* ================================================================ 2. LO QUE LA VITRINA MANDA (pedidos) ================================================================ */ function doPost(e) { var lock = LockService.getScriptLock(); lock.waitLock(20000); try { var d = e.parameter; obtenerHoja(HOJA_PEDIDOS, ENC_PEDIDOS).appendRow([ new Date(), d.jornada || '', d.nombre || '', "'" + (d.telefono || ''), d.punto || '', d.franja || '', Number(d.kg_total) || 0, Number(d.total) || 0, d.detalle || '', d.notas || '', 'Confirmado', 'No' ]); guardarLineas(d); consolidar(); return json({ ok: true }); } catch (err) { return json({ ok: false, error: String(err) }); } finally { lock.releaseLock(); } } function guardarLineas(d) { var lineas; try { lineas = JSON.parse(d.items || '[]'); } catch (x) { return; } if (!lineas.length) return; var hoja = obtenerHoja(HOJA_LINEAS, ENC_LINEAS); var ahora = new Date(); var filas = lineas.map(function (l) { return [ahora, d.jornada || '', d.nombre || '', l.producto, Number(l.kg), Number(l.precio), Number(l.subtotal)]; }); hoja.getRange(hoja.getLastRow() + 1, 1, filas.length, ENC_LINEAS.length).setValues(filas); } /* ================================================================ 3. LA LISTA DE COMPRA ================================================================ */ function consolidar() { var origen = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJA_LINEAS); if (!origen || origen.getLastRow() < 2) return; var jornada = jornadaVigente(); var suma = {}, clientes = {}; origen.getRange(2, 1, origen.getLastRow() - 1, ENC_LINEAS.length).getValues().forEach(function (f) { if (String(f[1]) !== jornada) return; var p = f[3]; if (!suma[p]) { suma[p] = { kg: 0, plata: 0, precio: f[5] }; clientes[p] = {}; } suma[p].kg += Number(f[4]); suma[p].plata += Number(f[6]); clientes[p][f[2]] = true; }); var hoja = obtenerHoja(HOJA_CONSOLIDADO, ['Producto', 'Kg a comprar', 'Clientes', 'Precio venta $/kg', 'Venta esperada $', 'Costo finca real $/kg', 'Costo total $']); if (hoja.getLastRow() > 1) hoja.getRange(2, 1, hoja.getLastRow() - 1, 7).clearContent(); var filas = Object.keys(suma) .sort(function (a, b) { return suma[b].kg - suma[a].kg; }) .map(function (p, i) { var r = i + 2; return [p, suma[p].kg, Object.keys(clientes[p]).length, suma[p].precio, suma[p].plata, '', '=IF(F' + r + '="","",B' + r + '*F' + r + ')']; }); if (!filas.length) return; hoja.getRange(2, 1, filas.length, 7).setValues(filas); var t = filas.length + 2; hoja.getRange(t, 1).setValue('TOTAL JORNADA ' + jornada); hoja.getRange(t, 2).setFormula('=SUM(B2:B' + (t - 1) + ')'); hoja.getRange(t, 5).setFormula('=SUM(E2:E' + (t - 1) + ')'); hoja.getRange(t, 7).setFormula('=SUM(G2:G' + (t - 1) + ')'); hoja.getRange(t, 1, 1, 7).setFontWeight('bold').setBackground('#E5D7B8'); hoja.autoResizeColumns(1, 7); } function jornadaVigente() { var hoy = new Date(), cierre = new Date(); cierre.setDate(hoy.getDate() + ((4 - hoy.getDay() + 7) % 7)); cierre.setHours(20, 0, 0, 0); if (cierre <= hoy) cierre.setDate(cierre.getDate() + 7); var sab = new Date(cierre); sab.setDate(cierre.getDate() + 2); return Utilities.formatDate(sab, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } /* ================================================================ 4. ARMADO INICIAL ================================================================ */ function crearHojasBase() { var ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss.getSheetByName(HOJA_CATALOGO)) { var cat = obtenerHoja(HOJA_CATALOGO, ENC_CATALOGO); cat.getRange(2, 1, 4, ENC_CATALOGO.length).setValues([ [1, 'Plátano hartón', 1900, 2200, 'Auto', 1, 'Verde o pintón, como lo pidas', 'fotos/platano.jpg', 'Si'], [2, 'Yuca', 1750, 2000, 'Auto', 1, 'Recién sacada, para el mismo día', 'fotos/yuca.jpg', 'Si'], [3, 'Papa pastusa', 2100, 2600, 'Si', 1, 'Bulto abierto, escogida', '', 'Si'], [4, 'Limón Tahití', 2350, 2400, 'No', 0.5, 'Jugoso, de cáscara delgada', '', 'Si'] ]); var reglaSiNo = SpreadsheetApp.newDataValidation() .requireValueInList(['Si', 'No'], true).setAllowInvalid(false).build(); var reglaAhorro = SpreadsheetApp.newDataValidation() .requireValueInList(['Auto', 'Si', 'No'], true).setAllowInvalid(false).build(); cat.getRange(2, 9, 200).setDataValidation(reglaSiNo); cat.getRange(2, 5, 200).setDataValidation(reglaAhorro); cat.getRange(2, 3, 200, 2).setNumberFormat('$#,##0'); cat.setColumnWidth(2, 170); cat.setColumnWidth(7, 260); cat.setColumnWidth(8, 170); cat.getRange(1, 11).setValue('CÓMO USAR ESTA HOJA').setFontWeight('bold'); cat.getRange(2, 11).setValue( 'Orden: en qué posición aparece en la vitrina (1 primero).\n' + 'Precio venta: el que ve el cliente, por kilo.\n' + 'Precio de referencia: lo que cuesta en la plaza. Alimenta el tachado y el "ahorras X%".\n' + 'Mostrar ahorro: Auto = solo si supera el 8%. Si = siempre. No = nunca.\n' + 'Paso: de a cuánto suma el botón +. Usa 0,5 en lo caro y 1 en lo pesado.\n' + 'Archivo de foto: la ruta dentro de tu repositorio, ej. fotos/platano.jpg. Vacío = se muestra la inicial.\n' + 'Disponible: pon No y el producto sale de la vitrina sin borrar la fila.\n\n' + 'Los cambios se ven apenas el cliente recargue. No hay que volver a publicar nada.' ).setWrap(true); cat.setColumnWidth(11, 460); } if (!ss.getSheetByName(HOJA_CONFIG)) { var cfg = obtenerHoja(HOJA_CONFIG, ['Parámetro', 'Valor']); cfg.getRange(2, 1, 7, 2).setValues([ ['Titular', 'Despensa del Campo'], ['Mensaje de la semana', 'Plátano, yuca y fruta compradas en finca el mismo sábado.'], ['Punto de recogida', 'Parqueadero Porto Vento — Girón'], ['Punto de recogida', 'Parque principal de Floridablanca'], ['Franja de entrega', 'Sábado 8:00 a 11:00 a.m.'], ['Franja de entrega', 'Sábado 3:00 a 6:00 p.m.'], ['Pedido mínimo', 20000] ]); cfg.setColumnWidth(1, 200); cfg.setColumnWidth(2, 380); cfg.getRange(1, 4).setValue( 'Puedes repetir "Punto de recogida" y "Franja de entrega" en varias filas: todas aparecen en la vitrina.' ).setWrap(true); cfg.setColumnWidth(4, 380); } obtenerHoja(HOJA_PEDIDOS, ENC_PEDIDOS); obtenerHoja(HOJA_LINEAS, ENC_LINEAS); SpreadsheetApp.getUi().alert('Listo. Ya puedes editar las hojas Catalogo y Config.'); } function obtenerHoja(nombre, encabezados) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var hoja = ss.getSheetByName(nombre); if (!hoja) { hoja = ss.insertSheet(nombre); hoja.appendRow(encabezados); hoja.getRange(1, 1, 1, encabezados.length) .setFontWeight('bold').setBackground('#17492C').setFontColor('#FAF5E9'); hoja.setFrozenRows(1); } return hoja; } /* ================================================================ 5. MENÚ ================================================================ */ function onOpen() { SpreadsheetApp.getUi().createMenu('Despensa') .addItem('Crear hojas base', 'crearHojasBase') .addSeparator() .addItem('Actualizar lista de compra', 'consolidar') .addItem('Ver resumen de la jornada', 'resumen') .addItem('Revisar el catálogo', 'revisarCatalogo') .addToUi(); } function resumen() { consolidar(); var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJA_CONSOLIDADO); if (!hoja || hoja.getLastRow() < 2) { SpreadsheetApp.getUi().alert('Todavía no hay pedidos para esta jornada.'); return; } var u = hoja.getLastRow(); SpreadsheetApp.getUi().alert( 'Jornada ' + jornadaVigente() + '\n\n' + 'Kilos a comprar: ' + hoja.getRange(u, 2).getValue() + ' kg\n' + 'Venta esperada: $' + Number(hoja.getRange(u, 5).getValue()).toLocaleString('es-CO') + '\n\n' + 'Si los kilos quedan por debajo de tu punto de equilibrio, el flete se come el margen.' ); } /** Avisa de errores del catálogo antes de que los vea un cliente. */ function revisarCatalogo() { var lista = leerCatalogo(), avisos = []; if (!lista.length) avisos.push('· El catálogo está vacío o ningún producto tiene precio.'); lista.forEach(function (p) { if (!p.hay) return; if (p.plaza && p.precio >= p.plaza) avisos.push('· ' + p.nombre + ': tu precio no es menor al de referencia.'); if (p.ahorro === 'si' && p.plaza && (1 - p.precio / p.plaza) < 0.05) avisos.push('· ' + p.nombre + ': estás forzando "ahorras" con menos del 5%. El cliente no lo siente.'); if (!p.nota) avisos.push('· ' + p.nombre + ': sin descripción. Una línea vende más que el nombre solo.'); }); var activos = lista.filter(function (p) { return p.hay; }).length; SpreadsheetApp.getUi().alert( 'Productos publicados: ' + activos + ' de ' + lista.length + '\n\n' + (avisos.length ? avisos.join('\n') : 'Sin observaciones. El catálogo está listo.') ); }
